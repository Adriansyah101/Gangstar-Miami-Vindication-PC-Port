dev
ini FULLSCREEN=1
ini SCREEN_WIDTH=1280
ini SCREEN_HEIGHT=720
ini SHORT_WARNING=1

key ENTER

wait 3

key ENTER
release ALT
wait 3
shot fs-city
expectlog ^\[gl\] window: full screen

shot fs-turned
